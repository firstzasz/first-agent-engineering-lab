+import unittest
+
+from relayboard.testing import request
+from relayboard.web import RelayBoardApp
+
+
+class FailureAlertTests(unittest.TestCase):
+    def make_app(self, max_attempts):
+        app = RelayBoardApp()
+        app.store.add_job("report", "Report", max_attempts=max_attempts)
+        self.addCleanup(app.store.connection.close)
+        return app
+
+    def start_run(self, app, source):
+        code, payload = request(
+            app, "POST", "/api/jobs/report/runs", {"source": source}
+        )
+        self.assertEqual(code, 201)
+        run_id = payload["run"]["id"]
+        self.assertEqual(payload["run"], {
+            "id": run_id, "job_id": "report",
+            "source": source, "status": "running",
+        })
+        return run_id
+
+    def attempt(self, app, run_id, source, succeeded, expected_status):
+        code, payload = request(
+            app, "POST", f"/api/runs/{run_id}/attempts",
+            {"succeeded": succeeded},
+        )
+        self.assertEqual(code, 200)
+        self.assertEqual(payload, {
+            "status": expected_status,
+            "run": {
+                "id": run_id, "job_id": "report", "source": source,
+                "status": "running" if expected_status == "retry" else expected_status,
+            },
+        })
+
+    def alerts(self, app):
+        code, payload = request(app, "GET", "/api/alerts")
+        self.assertEqual(code, 200)
+        return payload["alerts"]
+
+    def assert_attempts(self, app, run_id, outcomes):
+        self.assertEqual(
+            [(a.run_id, a.number, a.succeeded) for a in app.store.list_attempts(run_id)],
+            [(run_id, number, outcome) for number, outcome in enumerate(outcomes, 1)],
+        )
+
+    def assert_failure_alerts(self, app, run_ids):
+        self.assertEqual(self.alerts(app), [
+            {
+                "id": number, "job_id": "report", "run_id": run_id,
+                "kind": "run_failed",
+                "message": f"Job Report failed for run {run_id}",
+            }
+            for number, run_id in enumerate(run_ids, 1)
+        ])
+
+    def test_exhausted_retries_alert_only_at_terminal_failure(self):
+        for source in ("manual", "scheduled"):
+            for max_attempts in (1, 2, 3):
+                with self.subTest(source=source, max_attempts=max_attempts):
+                    app = self.make_app(max_attempts)
+                    run_id = self.start_run(app, source)
+                    alert_counts = []
+                    for number in range(1, max_attempts + 1):
+                        status = "retry" if number < max_attempts else "failed"
+                        self.attempt(app, run_id, source, False, status)
+                        self.assert_attempts(app, run_id, [False] * number)
+                        alert_counts.append(len(self.alerts(app)))
+                    self.assertEqual(alert_counts, [0] * (max_attempts - 1) + [1])
+                    self.assert_failure_alerts(app, [run_id])
+
+    def test_successful_retry_has_no_failure_alert_with_failure_control(self):
+        for source in ("manual", "scheduled"):
+            for max_attempts in (2, 3):
+                with self.subTest(source=source, max_attempts=max_attempts):
+                    app = self.make_app(max_attempts)
+                    run_id = self.start_run(app, source)
+                    alert_counts = []
+                    for _ in range(max_attempts - 1):
+                        self.attempt(app, run_id, source, False, "retry")
+                        alert_counts.append(len(self.alerts(app)))
+                    self.attempt(app, run_id, source, True, "succeeded")
+                    self.assert_attempts(app, run_id, [False] * (max_attempts - 1) + [True])
+                    alert_counts.append(len(self.alerts(app)))
+
+                    failed_run_id = self.start_run(app, source)
+                    for number in range(1, max_attempts + 1):
+                        status = "retry" if number < max_attempts else "failed"
+                        self.attempt(app, failed_run_id, source, False, status)
+                    self.assert_attempts(app, failed_run_id, [False] * max_attempts)
+                    self.assert_failure_alerts(app, [failed_run_id])
+                    self.assertEqual(alert_counts, [0] * max_attempts)
+
+    def test_distinct_failed_runs_of_same_job_each_keep_their_alert(self):
+        for max_attempts in (1, 2, 3):
+            with self.subTest(max_attempts=max_attempts):
+                app = self.make_app(max_attempts)
+                run_ids = []
+                for source in ("manual", "scheduled"):
+                    run_id = self.start_run(app, source)
+                    run_ids.append(run_id)
+                    for number in range(1, max_attempts + 1):
+                        status = "retry" if number < max_attempts else "failed"
+                        self.attempt(app, run_id, source, False, status)
+                    self.assert_attempts(app, run_id, [False] * max_attempts)
+                self.assertNotEqual(run_ids[0], run_ids[1])
+                self.assert_failure_alerts(app, run_ids)
+
+
+if __name__ == "__main__":
+    unittest.main()
